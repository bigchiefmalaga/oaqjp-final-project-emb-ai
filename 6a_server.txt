from flask import Flask, request, render_template

import sys, json
sys.path.append('./oaqjp-final-project-emb-ai')  # Chemin précis

from emotion_detection import emotion_detector

app = Flask(__name__, template_folder='oaqjp-final-project-emb-ai/templates', static_folder='oaqjp-final-project-emb-ai/static')

@app.route('/')
def index():
    return render_template('index.html')

@app.route('/emotionDetector', methods=['POST','GET'])
def flask_handler():
    if request.method == 'GET':
        query = request.args.get('textToAnalyze')
    else: 
        query = request.form.get('textToAnalyze')
    if not query:
        return "No text provided", 400
    result = emotion_detector(query)
    resultjson = json.dumps(result,indent=2)
    emotions_text = (f"For the given statement, the system response is 'anger': {result['anger']}, "
                 f"'disgust': {result['disgust']}, 'fear': {result['fear']}, "
                 f"'joy': {result['joy']} and 'sadness': {result['sadness']}. "
                 f"The dominant emotion is **{result['dominant_emotion']}**.")
    return emotions_text

if __name__ == '__main__':
    app.run(debug=True, host='0.0.0.0', port=5000)
